# kbench.suprx (ARM). cmake -DADRENALINE_SRC=<raiz de Adrenaline> ..   (o -DKB_MODE=<n>)
cmake_minimum_required(VERSION 3.16)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain")
endif()
project(kbench C)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -O2 -Wall -nostdlib -fno-builtin")
include_directories(../comun)
if(ADRENALINE_SRC)  # raiz de isage/Adrenaline: KERMIT_MODE_EXTRA_1 de adrenaline_vita.h
  add_definitions(-DKB_USE_ADRENALINE_H)
  include_directories(${ADRENALINE_SRC})
elseif(KB_MODE)
  add_definitions(-DKB_MODE=${KB_MODE})
endif()
add_executable(kbench main.c ../comun/walk.c)
target_link_libraries(kbench SceCompat_stub SceKernelDmacMgr_stub SceLibKernel_stub SceKernelThreadMgr_stub SceKernelModulemgr_stub
  SceSysmem_stub SceProcessmgr_stub SceIofilemgr_stub ScePower_stub gcc)
vita_create_self(kbench.suprx kbench CONFIG exports.yml UNSAFE)
